  111| async function pedirAudio(base, t, emocao, vel, tom, limite = 5000) {
  112|   const chave = `${emocao}|${vel}|${tom}|${t}`;
  113|   if (guardadas.has(chave)) return guardadas.get(chave).slice(0);
  114|   const r = await fetch(`${base}/voz`, {
  115|     method: 'POST', headers: { 'content-type': 'application/json' }, signal: tempoLimite(limite),
  116|     body: JSON.stringify({ t, r: vel, p: tom, e: EMOCAO_DA_VOZ[emocao] || 'calmo' }),
  117|   });
  118|   if (!r.ok || !/audio/.test(r.headers.get('content-type') || '')) {
  119|     const j = await r.json().catch(() => ({}));
  120|     throw Object.assign(new Error(`voz ${r.status}`), { status: r.status, detalhe: j });
  121|   }
  122|   ultimaVoz = r.headers.get('x-voz') || '';
  123|   const buf = await r.arrayBuffer();
  124|   guardadas.set(chave, buf.slice(0));
  125|   if (guardadas.size > 50) guardadas.delete(guardadas.keys().next().value);
  126|   return buf;
  127| }

  151| // toca o MP3 e mede o volume enquanto toca (é isso que mexe a boca do avatar no ritmo da fala)
  152| async function tocar(buf, minha) {
  153|   const c = audioCtx();
  154|   if (!c) throw new Error('sem áudio');
  155|   // iPhone: depois de ir pro fundo, o áudio fica "interrupted" (não só "suspended"); tenta voltar antes de tocar
  156|   if (c.state !== 'running') { await c.resume().catch(() => {}); if (c.state !== 'running') throw new Error('áudio travado'); }
  157|   const dados = await c.decodeAudioData(buf);
  158|   if (minha !== corte) return;
  159|   const s = c.createBufferSource();
  160|   s.buffer = dados;
  161|   const an = c.createAnalyser();
  162|   an.fftSize = 512;
  163|   s.connect(an); an.connect(c.destination);
  164|   fonte = s;
  165|   const amostra = new Uint8Array(an.fftSize);
  166|   let raf = 0;
  167|   const medir = () => {
  168|     an.getByteTimeDomainData(amostra);
  169|     let soma = 0;
  170|     for (const v of amostra) { const x = (v - 128) / 128; soma += x * x; }
  171|     avisar('nivel', Math.min(1, Math.sqrt(soma / amostra.length) * 4));
  172|     raf = requestAnimationFrame(medir);
  173|   };
  174|   await new Promise(fim => {
  175|     s.onended = () => { cancelAnimationFrame(raf); if (fonte === s) fonte = null; fim(); };
  176|     avisar('inicio', 'nuvem');
  177|     s.start(0);
  178|     raf = requestAnimationFrame(medir);
  179|   });
  180| }

  182| function falarNoAparelho(t, emocao, minha) {
  183|   if (!vozDisponivel() || minha !== corte) return Promise.resolve();
  184|   return new Promise(fim => {
  185|     try {
  186|       const u = new SpeechSynthesisUtterance(t);
  187|       u.lang = 'pt-BR';
  188|       const v = melhorVoz();
  189|       if (v) u.voice = v;
  190|       const [vel, tom] = JEITO[emocao] || JEITO.calmo;
  191|       // sem voz masculina no aparelho, ele fala um pouco mais grave (como no Perfil)
  192|       u.pitch = Math.max(0.4, Math.min(2, (v && !MASC.test(v.name || '') ? 0.8 : 1) * tom));
  193|       u.rate = Math.min(2, Math.max(0.5, vel * VELOCIDADE));
  194|       const acabou = () => { clearTimeout(tempo); fim(); };
  195|       const tempo = setTimeout(acabou, Math.min(30_000, 1500 + t.length * 90 / u.rate));
  196|       u.onboundary = () => avisar('palavra'); // cada palavra: a boca dá uma abridinha
  197|       u.onend = u.onerror = acabou;
  198|       speechSynthesis.speak(u);
  199|       // 17/11 (JF: "os avatares não estão mexendo a boca"): no app instalado no iPhone o aviso de "começou" (onstart)
  200|       // nem sempre chega; a boca começa junto com o pedido de fala, sem depender dele
  201|       avisar('inicio', 'aparelho');
  202|     } catch { fim(); }
  203|   });
  204| }

  206| async function falarAgora(t, emocao, minha) {
  207|   const base = servidorDaVoz();

  209|   try {
  210|     if (base && (typeof navigator === 'undefined' || navigator.onLine !== false) && await nuvemLigada(base)) {
  211|       try {
  212|         const buf = await pedirAudio(base, t, emocao, Math.round(vel * VELOCIDADE * 20) / 20, Math.round(tom * 20) / 20);
  213|         if (minha !== corte) return;
  214|         await tocar(buf, minha);
  215|         return;
  216|       } catch { /* demorou, falhou ou o som está travado: a voz do aparelho fala */ }
  217|     }
  218|     if (minha === corte) await falarNoAparelho(t, emocao, minha);
  219|   } finally {
  220|     if (minha === corte) avisar('fim');
  221|   }
  222| }

  298| export function ouvir({ limiteMs = 12000 } = {}) {
  299|   return new Promise((resolve, reject) => {
  300|     const R = window.SpeechRecognition || window.webkitSpeechRecognition;
  301|     if (!R) return reject(new Error('Ditado não disponível neste navegador.'));
  302|     let r, acabou = false, texto = '';
  303|     const fim = (erro) => {
  304|       if (acabou) return;
  305|       acabou = true;
  306|       clearTimeout(relogio);
  307|       atual = null;
  308|       if (texto) resolve(texto);
  309|       else reject(Object.assign(new Error(erro || 'Não consegui ouvir.'), { semResultado: true }));
  310|     };
  311|     try {
  312|       r = new R();
  313|       r.lang = 'pt-BR';
  314|       r.interimResults = false;
  315|       r.maxAlternatives = 1;
  316|       r.continuous = false;
  317|       r.onresult = e => { texto = (e.results?.[0]?.[0]?.transcript || '').trim(); if (texto) fim(); };
  318|       r.onerror = e => fim(MOTIVO[e.error] || `Não consegui ouvir (${e.error || 'erro desconhecido'}).`);
  319|       r.onnomatch = () => fim('Não entendi o áudio. Tenta de novo, mais devagar.');
  320|       r.onend = () => fim(texto ? null : 'O navegador parou de ouvir sem me passar o que você disse.');
  321|       atual = r;
  322|       r.start();
  323|     } catch (e) {
  324|       return fim(`Não consegui ligar o microfone (${e.message || e}).`);
  325|     }
  326|     const relogio = setTimeout(() => { try { r.stop(); } catch {} fim('Demorou demais e eu parei de ouvir.'); }, limiteMs);
  327|   });
  328| }

   38| /** Texto já limpo (sem < > & ") → SSML, igual ao Perfil. */
   39| export function vozSsml(nome, t, rateN, pitch, e, estilos) {
   40|   const [vol, pausa, estilo, dRate] = VOZ_EMOCOES[e] || VOZ_EMOCOES.calmo;
   41|   const rn = Math.round((Number(rateN) || 0) + (dRate || 0)), rate = (rn >= 0 ? '+' : '') + rn + '%';
   42|   if (!/[.!?…]$/.test(t)) t += '.'; // sem ponto no fim, a voz às vezes "solta o ar"
   43|   let corpo = t
   44|     .replace(/^(ah+|hm+|hum+|u[eé]|ops|opa|uau|eita|nossa|vish|oxe|aff)([,!.]?)\s+(?!⏸)/i, (m, a, pt) => `${a}${pt || ','}<break time="220ms"/> `)
   45|     .replace(/\s*⏸\s*/g, `<break time="${pausa}ms"/> `)
   46|     .replace(/\p{L}*(\p{L})\1{2,}\p{L}*/gu, w => `<prosody rate="-15%" volume="+20%">${w.replace(/(\p{L})\1{2,}/gu, '$1').toLowerCase()}</prosody>`)
   47|     .replace(/(^|[^\p{L}])(\p{Lu}{4,})(?![\p{L}])/gu, (m, a, w) => `${a}<prosody volume="+25%"${estilos ? '' : ' pitch="+8%"'}>${w.toLowerCase()}</prosody>`);
   48|   corpo = `<prosody rate="${rate}"${estilos ? '' : ` pitch="${pitch}"`} volume="${vol}">${corpo}</prosody>`;
   49|   if (estilos && estilo) corpo = `<mstts:express-as style="${estilo}">${corpo}</mstts:express-as>`;
   50|   return `<speak version="1.0" xmlns="http://www.w3.org/2001/10/synthesis" xmlns:mstts="http://www.w3.org/2001/mstts" xml:lang="pt-BR"><voice name="${nome}">${corpo}</voice></speak>`;
   51| }

   75| /** POST /voz { t, r, p, e } → MP3 (ou JSON de erro; aí o app fala com a voz do aparelho). */
   76| export async function vozFalar(req, env, h = {}) {
   77|   const vozes = vozesDoServidor(env);
   78|   if (!vozes.length) return json({ erro: 'Voz natural não configurada no servidor.' }, 503, h);
   79|   let d;
   80|   try {
   81|     const bruto = await req.text();
   82|     if (bruto.length > 4000) return json({ erro: 'grande' }, 413, h);
   83|     d = JSON.parse(bruto);
   84|   } catch {
   85|     return json({ erro: 'json' }, 400, h);
   86|   }
   87|   const t = String(d?.t || '').replace(/[\u0000-\u001f<>&"]/g, ' ').replace(/\s+/g, ' ').trim().slice(0, 400);
   88|   if (!t) return json({ erro: 'vazio' }, 400, h);
   89|   const num = (v, a, b) => Math.max(a, Math.min(b, Number.isFinite(+v) ? +v : 1));
   90|   const r = Math.round(num(d.r, 0.5, 2) * 20) / 20, p = Math.round(num(d.p, 0.5, 2) * 20) / 20;
   91|   const e = Object.prototype.hasOwnProperty.call(VOZ_EMOCOES, d.e) ? d.e : '';
   92|   const kv = env.CARTEIRA_KV;
   93|   // x-voz exposto (17/11): o "Testar a voz" do app mostra qual voz respondeu
   94|   const mp3 = (dados, origem) => new Response(dados, { headers: { 'content-type': 'audio/mpeg', 'cache-control': 'public, max-age=604800', 'x-voz': origem, 'access-control-expose-headers': 'x-voz', ...h } });
   95|   const chaveDe = async v => {
   96|     const dig = await crypto.subtle.digest('SHA-256', new TextEncoder().encode([v.nome, r, p, t, e, v.estilos ? 1 : 0, VOZ_SSML_VERSAO].join('|')));
   97|     return 'voz:' + [...new Uint8Array(dig)].map(b => b.toString(16).padStart(2, '0')).join('').slice(0, 40);
   98|   };
   99|   const agora = Date.now();
  100|   const pausa = Number(env.VOZ_PAUSA_MS) > 0 ? Number(env.VOZ_PAUSA_MS) : 600e3;
  101|   const ativas = vozes.filter((v, i) => i === vozes.length - 1 || !(agora - (vozFora[v.id] || 0) < pausa));
  102|   // a mesma fala, do mesmo jeito, volta do cache (as falas se repetem muito: não gasta o Azure de novo)
  103|   if (kv) for (const v of ativas) {
  104|     const g = await kv.get(await chaveDe(v), 'arrayBuffer');
  105|     if (g) return mp3(g, 'guardada');
  106|   }
  107|   const ip = req.headers.get('cf-connecting-ip') || 'x';
  108|   const rv = (ritmoVoz.get(ip) || []).filter(x => agora - x < 3600e3);
  109|   if (rv.length >= 400) return json({ erro: 'calma' }, 429, h);
  110|   rv.push(agora);
  111|   ritmoVoz.set(ip, rv);
  112|   // teto do mês em letras (padrão 100 mil: com os 400 mil do Perfil, cabe nas 500 mil grátis do Azure, se a conta for a mesma)
  113|   const mes = new Date().toISOString().slice(0, 7);
  114|   const teto = Number(env.VOZ_TETO_MES) > 0 ? Number(env.VOZ_TETO_MES) : 100000;
  115|   const gasto = kv ? Number(await kv.get(`voz:mes:${mes}`)) || 0 : 0;
  116|   if (gasto + t.length > teto) return json({ erro: 'teto' }, 429, h);
  117|   const pct = (x, k) => (x >= 1 ? '+' : '') + Math.round((x - 1) * k) + '%';
  118|   let ultimo = 0;
  119|   for (const v of ativas) {
  120|     let az = null;
  121|     try {
  122|       az = await fetch(v.url, {
  123|         method: 'POST',
  124|         headers: { 'Ocp-Apim-Subscription-Key': v.chave, 'Content-Type': 'application/ssml+xml', 'X-Microsoft-OutputFormat': 'audio-24khz-48kbitrate-mono-mp3', 'User-Agent': 'carteira-jf' },
  125|         body: vozSsml(v.nome, t, Math.round((r - 1) * 100), pct(p, 40), e, v.estilos),
  126|       });
  127|     } catch { az = null; }
  128|     if (!az || !az.ok) {
  129|       ultimo = az ? az.status : 0;
  130|       vozFora[v.id] = agora;
  131|       continue;
  132|     }
  133|     const dados = await az.arrayBuffer();
  134|     if (kv) {
  135|       await kv.put(`voz:mes:${mes}`, String(gasto + t.length), { expirationTtl: 40 * 86400 });
  136|       if (dados.byteLength > 0 && dados.byteLength < 1900000) await kv.put(await chaveDe(v), dados, { expirationTtl: 30 * 86400 });
  137|     }
  138|     return mp3(dados, 'nova:' + v.id);
  139|   }
  140|   return json({ erro: 'azure', status: ultimo }, 502, h);
  141| }
