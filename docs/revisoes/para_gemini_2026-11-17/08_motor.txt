   35|  * Texto digitado -> centavos inteiros. Aceita "12", "12,5", "12,50", "1.234,56", "R$ 9,90", "12.50".

   38| export function parseAmount(input) {
   39|   if (typeof input === 'number') return Number.isFinite(input) && input > 0 ? Math.round(input * 100) : null;
   40|   if (typeof input !== 'string') return null;
   41|   let s = input.replace(/R\$|\s/gi, '');
   42|   if (!/^[\d.,]+$/.test(s)) return null;
   43|   if (s.includes(',')) {
   44|     s = s.replace(/\./g, '').replace(',', '.');
   45|   } else if ((s.match(/\./g) || []).length > 1 || /\.\d{3}$/.test(s)) {
   46|     s = s.replace(/\./g, ''); // "1.234" = mil e pouco, não 1,234
   47|   }
   48|   const n = Number(s);
   49|   if (!Number.isFinite(n) || n <= 0) return null;

  208| /**
  209|  * Dias seguidos em que a pessoa usou o app pra anotar (gasto ou entrada criados naquele dia, ou "hoje não gastei").
  210|  * A ofensiva continua viva se o último dia foi hoje ou ontem. Devolve { dias, hojeFeito, recorde }.
  211|  */
  212| function diasAnotados({ expenses = [], incomes = [], diasSemGasto = [] }) {
  213|   const dias = new Set();
  214|   for (const x of [...expenses, ...incomes]) if (Number.isFinite(x.createdAt)) dias.add(diaLocal(x.createdAt));
  215|   for (const d of diasSemGasto) if (isValidISODate(d)) dias.add(d);
  216|   return dias;
  217| }
  218| 
  219| export function ofensiva(dados, today = toISODate()) {
  220|   const dias = diasAnotados(dados);
  221|   const hojeFeito = dias.has(today);
  222|   let cursor = hojeFeito ? today : diaAnterior(today);
  223|   let atual = 0;
  224|   while (dias.has(cursor)) { atual++; cursor = diaAnterior(cursor); }
  225|   // recorde: maior sequência em todo o histórico
  226|   let recorde = 0;
  227|   for (const d of dias) {
  228|     if (dias.has(diaAnterior(d))) continue; // só começa a contar no primeiro dia de cada sequência
  229|     let n = 0, c = d;
  230|     while (dias.has(c)) { n++; const [y, m, dd] = c.split('-').map(Number); c = toISODate(new Date(y, m - 1, dd + 1)); }
  231|     if (n > recorde) recorde = n;
  232|   }
  233|   return { dias: atual, hojeFeito, recorde: Math.max(recorde, atual) };
  234| }

  249| /**
  250|  * Fato útil da semana (15/11, no lugar do troféu da ofensiva; ideia do Code): compara os últimos 7 dias com os 7
  251|  * anteriores. Só fala se as duas semanas tiverem gasto anotado em pelo menos 3 dias e a diferença for de R$ 10 ou mais.
  252|  * Devolve { type: 'semana_menos' | 'semana_mais', amountCents (diferença), totalCents (esta semana),
  253|  *           categoryName, overCents (quanto a categoria que mais mudou variou) } ou null.
  254|  */
  255| export function fatoDaSemana(expenses, categories = [], today = toISODate()) {
  256|   const [y, m, d] = today.split('-').map(Number);
  257|   const dia = n => toISODate(new Date(y, m - 1, d - n));
  258|   const ini = dia(6), iniAntes = dia(13), fimAntes = dia(7);
  259|   const desta = expenses.filter(e => e.date >= ini && e.date <= today);
  260|   const antes = expenses.filter(e => e.date >= iniAntes && e.date <= fimAntes);
  261|   if (new Set(desta.map(e => e.date)).size < 3 || new Set(antes.map(e => e.date)).size < 3) return null;
  262|   const soma = l => l.reduce((t, e) => t + e.amountCents, 0);
  263|   const diff = soma(desta) - soma(antes);
  264|   if (Math.abs(diff) < 1000) return null;
  265|   const porCat = new Map();
  266|   for (const e of desta) porCat.set(e.categoryId, (porCat.get(e.categoryId) || 0) + e.amountCents);
  267|   for (const e of antes) porCat.set(e.categoryId, (porCat.get(e.categoryId) || 0) - e.amountCents);
  268|   // a categoria que mais ajudou (semana menor) ou mais pesou (semana maior)
  269|   const [catId, catDiff] = [...porCat].sort((a, b) => (diff < 0 ? a[1] - b[1] : b[1] - a[1]))[0] || [];
  270|   const cat = categories.find(c => c.id === catId);
  271|   return {
  272|     type: diff < 0 ? 'semana_menos' : 'semana_mais', amountCents: Math.abs(diff), totalCents: soma(desta),
  273|     categoryName: cat ? cat.name : 'outras coisas', overCents: catDiff ? Math.abs(catDiff) : 0,
  274|   };
  275| }

  321| /**
  322|  * Eventos disparados por um gasto recém-cadastrado.
  323|  * `expenses` já inclui `newExpense`. Retorna lista de eventos (só leitura para o narrador).
  324|  */
  325| export function detectExpenseEvents({ expenses, categories, newExpense }) {
  326|   const cat = categories.find(c => c.id === newExpense.categoryId);
  327|   const events = [{
  328|     type: 'gasto_cadastrado',
  329|     categoryId: newExpense.categoryId,
  330|     categoryName: cat?.name ?? newExpense.categoryId,
  331|     amountCents: newExpense.amountCents,
  332|   }];
  333| 
  334|   const budget = cat?.budgetCents || 0;
  335|   if (budget > 0) {
  336|     const after = spentIn(expenses, newExpense.month, newExpense.categoryId);
  337|     const before = after - valorProprio(newExpense);
  338|     const base = {
  339|       categoryId: cat.id, categoryName: cat.name, spentCents: after, budgetCents: budget,
  340|       pct: Math.round((after / budget) * 100),
  341|     };
  342|     if (before <= budget && after > budget) {
  343|       events.push({ type: 'categoria_estourada', ...base, overCents: after - budget });
  344|     } else if (before < budget * ALERT_RATIO && after >= budget * ALERT_RATIO) {
  345|       events.push({ type: 'orcamento_80', ...base, leftCents: budget - after });
  346|     }
  347|   }
  348| 
  349|   if (newExpense.categoryId === 'delivery') {
  350|     const recent = expenses.filter(e => {
  351|       if (e.categoryId !== 'delivery') return false;
  352|       const diff = daysBetween(e.date, newExpense.date);
  353|       return diff >= 0 && diff < DELIVERY_WINDOW_DAYS;
  354|     });
  355|     if (recent.length >= DELIVERY_REPEAT_COUNT) {
  356|       events.push({
  357|         type: 'delivery_repetido',
  358|         count: recent.length,
  359|         days: DELIVERY_WINDOW_DAYS,
  360|         totalCents: recent.reduce((s, e) => s + valorProprio(e), 0),
  361|       });
  362|     }
  363|   }
  364|   // formiguinha (JF, 16/11): o 10º, 20º, 30º… gastinho do mês dispara o aviso (com o total, que é o que assusta)
  365|   const valorNovo = valorProprio(newExpense);
  366|   if (!newExpense.compromissoId && valorNovo > 0 && valorNovo <= FORMIGA_LIMITE_CENTS) {

  402|   if (!pagou) return [];

  594| export function resumoPisos(accounts, expenses, incomes, transferencias = [], today = toISODate()) {
  595|   const contas = accounts.filter(a => a.tipo !== 'cartao' && !a.paraQuem).map(a => {

 1623| /**
 1624|  * Mostra o efeito de uma compra na sobra prevista do mês. Só leitura: não grava nada.
 1625|  * `sobraCents` vem de `analisarPlanejamento` (null = sem entrada nem orçamento pra comparar).

 1628|  */
 1629| export function simularCompra({ valorCents, parcelas = 1 }, { sobraCents, baseCents = 0 }) {
 1630|   const n = Math.min(Math.max(Math.trunc(parcelas) || 1, 1), 24);
 1631|   const parcelaCents = Math.ceil(valorCents / n);
 1632|   if (sobraCents == null) return { situacao: 'sem_base', parcelaCents, parcelas: n };
 1633|   const sobraDepoisCents = sobraCents - parcelaCents;
 1634|   let situacao = 'folga';
 1635|   if (sobraDepoisCents < 0) situacao = 'nao_cabe';
 1636|   else if (sobraDepoisCents < Math.round(baseCents * 0.1)) situacao = 'aperta'; // menos de 10% da base de reserva
 1637|   // não coube: em quantas vezes (até 12) a parcela caberia na sobra?
 1638|   let parcelasQueCabem = null;
 1639|   if (situacao === 'nao_cabe' && sobraCents > 0) {
 1640|     const k = Math.ceil(valorCents / sobraCents);
 1641|     if (k <= 12 && k > n) parcelasQueCabem = k;
 1642|   }
 1643|   return { situacao, parcelaCents, parcelas: n, sobraDepoisCents, faltaCents: Math.max(0, -sobraDepoisCents), parcelasQueCabem };

 1695| // ---------- Máscara de dinheiro (JF, 13/11: "digitei 178,35 e saiu 17.835") ----------
 1696| /**
 1697|  * Como nos apps de banco: os dois últimos números são os centavos e o resto vai enchendo da direita pra esquerda.
 1698|  * 1 → "0,01", 17 → "0,17", 178 → "1,78", 17835 → "178,35", 123456 → "1.234,56". Vírgula e ponto digitados são
 1699|  * ignorados (só contam os números), então "178,35" colado também vira "178,35". `sinal`: aceita "-" na frente (saldo).
 1700|  */
 1701| export function mascaraCentavos(texto, { sinal = false } = {}) {
 1702|   const s = String(texto ?? '');
 1703|   const neg = sinal && /^\s*-/.test(s);
 1704|   const dig = s.replace(/\D/g, '').replace(/^0+/, '').slice(0, 13);
 1705|   if (!dig) return neg ? '-' : '';
 1706|   const c = Number(dig);
 1707|   const reais = Math.floor(c / 100).toLocaleString('pt-BR');
 1708|   return `${neg ? '-' : ''}${reais},${String(c % 100).padStart(2, '0')}`;
 1709| }
