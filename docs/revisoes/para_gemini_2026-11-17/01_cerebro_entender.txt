   84| /** "juros compostos de 1000 a 1% por 12 meses" → { tipo, valor, taxa (fração ao mês), meses }. */
   85| export function lerCalculoJuros(texto) {
   86|   const t = String(texto || '').normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase();
   87|   const PCT = '(?:%|porcento|por cento)';
   88|   const temPct = new RegExp(`\\d\\s*${PCT}`).test(t);
   89|   const tipo = /juros? compost/.test(t) ? 'compostos' : /juros? simples/.test(t) ? 'simples' : /(quanto (rende|renderia|vai render|fica)|se eu (guardar|investir|aplicar))/.test(t) && temPct ? 'compostos' : null;
   90|   if (!tipo) return null;
   91|   const taxa = t.match(new RegExp(`(\\d+(?:[.,]\\d+)?)\\s*${PCT}\\s*(ao|a|por)?\\s*(mes|ano|a\\.?m\\.?|a\\.?a\\.?)?`));
   92|   const prazo = t.match(/(\d+)\s*(meses|mes|anos|ano)\b/);
   93|   const semTaxa = t.replace(new RegExp(`(\\d+(?:[.,]\\d+)?)\\s*${PCT}`, 'g'), ' ').replace(/(\d+)\s*(meses|mes|anos|ano)\b/g, ' ');
   94|   const valor = acharValor(semTaxa);
   95|   if (!taxa || !prazo || !valor) return { tipo, faltando: true };
   96|   let i = Number(taxa[1].replace(',', '.')) / 100;
   97|   const anual = /ano|a\.?a/.test(taxa[3] || '');
   98|   let meses = Number(prazo[1]) * (/ano/.test(prazo[2]) ? 12 : 1);
   99|   // taxa ao ano: no composto vira a mensal equivalente; no simples, divide por 12
  100|   const iSimples = anual ? i / 12 : i;
  101|   const iComposto = anual ? Math.pow(1 + i, 1 / 12) - 1 : i;
  102|   if (meses > 600 || i > 1) return { tipo, faltando: true };
  103|   return { tipo, valor, taxa: tipo === 'simples' ? iSimples : iComposto, taxaSimples: iSimples, meses, taxaTexto: `${taxa[1]}% ${anual ? 'ao ano' : 'ao mês'}` };
  104| }

  193| /** Distância de edição com troca de letras vizinhas contando 1 ("gastie" ~ "gastei"). */
  194| function distancia(a, b, max) {
  195|   if (Math.abs(a.length - b.length) > max) return max + 1;
  196|   const d = Array.from({ length: a.length + 1 }, (_, i) => [i, ...Array(b.length).fill(0)]);
  197|   for (let j = 1; j <= b.length; j++) d[0][j] = j;
  198|   for (let i = 1; i <= a.length; i++) {
  199|     for (let j = 1; j <= b.length; j++) {
  200|       const c = a[i - 1] === b[j - 1] ? 0 : 1;
  201|       d[i][j] = Math.min(d[i - 1][j] + 1, d[i][j - 1] + 1, d[i - 1][j - 1] + c);
  202|       if (i > 1 && j > 1 && a[i - 1] === b[j - 2] && a[i - 2] === b[j - 1]) d[i][j] = Math.min(d[i][j], d[i - 2][j - 2] + 1);
  203|     }
  204|   }
  205|   return d[a.length][b.length];
  206| }

  208| export function corrigirDigitacao(t, extras = []) {
  209|   const vocab = [...VOCAB, ...extras];
  210|   return String(t).split(' ').map(w => {
  211|     if (w.length < 4 || VOCAB_SET.has(w) || extras.includes(w) || /\d/.test(w)) return w;
  212|     const max = w.length >= 8 ? 2 : 1;
  213|     let melhor = null;
  214|     for (const v of vocab) {
  215|       if (Math.abs(v.length - w.length) > max || v[0] !== w[0]) continue; // primeira letra quase nunca erra
  216|       // 4 letras tem vizinho demais (mesa/mes, home/hoje): aí só vale a letra que faltou ("sado" → "saldo") (16/11)
  217|       if (w.length === 4 && v.length <= w.length) continue;
  218|       const dd = distancia(w, v, max);
  219|       if (dd <= max && (!melhor || dd < melhor.d)) melhor = { v, d: dd };
  220|     }
  221|     return melhor ? melhor.v : w;
  222|   }).join(' ');

  851|   if (!desc) return desc;
  852|   desc = desc.replace(HORAS, ' ').replace(/\b(da manha|da tarde|da noite|da madrugada)\b/g, ' ').replace(/\b(dia \d{1,2}|anteontem|ontem|hoje)\b/gi, ' ').replace(/\bdia\s*$/i, ' ').replace(/\s+/g, ' ').trim();
  853|   if (!desc) return null;
  854|   const bancos = new Set(accounts.flatMap(a => [a.bank, a.nickname]).filter(Boolean).map(norm));
  855|   const LIGA = /^(no|na|de|do|da|em|a|com|pelo|pela|e)$/;
  856|   const palavras = desc.split(/\s+/);
  857|   // nome do banco com mais de uma palavra ("Banco Teste"): marca a sequência toda
  858|   const doBanco = new Set();
  859|   for (const b of bancos) {
  860|     const bt = b.split(/\s+/);
  861|     for (let i = 0; i + bt.length <= palavras.length; i++) {
  862|       if (bt.every((x, k) => norm(palavras[i + k]) === x)) bt.forEach((_, k) => doBanco.add(i + k));
  863|     }
  864|   }
  865|   const limpo = [];
  866|   for (const [i, w] of palavras.entries()) {
  867|     const n = norm(w);
  868|     // número solto só sai se for parcela ("em 3 vezes"); "no 99", "padaria 2 irmãos" ficam (achado do Gemini, 12/11)
