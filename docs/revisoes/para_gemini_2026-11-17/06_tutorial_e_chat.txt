  124| function teclas(e) {
  125|   if (!atual) return;
  126|   if (e.key === 'Escape') terminar(false);
  127|   else if (e.key === 'ArrowRight') irPasso(atual.i + 1, 1);
  128|   else if (e.key === 'ArrowLeft') irPasso(atual.i - 1, -1);

  131| async function irPasso(i, dir = 1) {
  132|   if (!atual) return;
  133|   if (i >= atual.passos.length) return terminar(true);
  134|   if (i < 0) return;
  135|   atual.i = i;
  136|   const p = atual.passos[i];
  137|   const cartao = $('#tour-cartao');
  138|   cartao.classList.add('trocando');
  139|   if (location.hash.slice(1) !== p.tela) location.hash = p.tela;
  140|   await espera(MENOS_MOVIMENTO.matches ? 120 : 380);
  141|   if (!atual) return;
  142|   const visivel = sel => { const x = sel && document.querySelector(sel); return x && x.getClientRects().length ? x : null; };
  143|   const alvoAlt = !visivel(p.alvo) && visivel(p.alvoAlt);
  144|   const alvo = visivel(p.alvo) || alvoAlt;
  145|   // pedaço que não existe ou está escondido (ex.: tela grande sem a barra de baixo): pula sem travar
  146|   if (!alvo || !alvo.getClientRects().length) {
  147|     return dir < 0 && i > 0 ? irPasso(i - 1, -1) : irPasso(i + 1, 1);
  148|   }
  149|   if (alvo.tagName === 'DETAILS') alvo.open = false; // fechado fica menor e cabe na tela
  150|   alvo.scrollIntoView({ block: 'center', behavior: MENOS_MOVIMENTO.matches ? 'auto' : 'smooth' });
  151|   await espera(MENOS_MOVIMENTO.matches ? 60 : 320);
  152|   if (!atual) return;
  153|   atual.alvo = alvo;
  154|   $('#tour-passo').textContent = `Passo ${p.n} de ${p.de}`;
  155|   $('#tour-tit').textContent = p.titulo;
  156|   $('#tour-txt').textContent = alvoAlt && p.textoAlt ? p.textoAlt : p.texto;
  157|   $('#tour-pontos').innerHTML = atual.passos.map((_, k) => `<i class="${k === i ? 'atual' : k < i ? 'feito' : ''}"></i>`).join('');
  158|   $('#tour-voltar').hidden = i === 0;
  159|   $('#tour-proximo').textContent = i === atual.passos.length - 1 ? 'Terminar ✓' : 'Próximo ›';
  160|   avatarTour?.mostrar(p.forma);
  161|   posicionar();
  162|   cartao.classList.remove('trocando');
  163|   $('#tour-proximo').focus({ preventScroll: true });
  164| }

  174|     left: `${Math.max(4, r.left - folga)}px`, top: `${Math.max(4, r.top - folga)}px`,

  419| // Teclado do iPhone (14/11, JF: "as barras descem junto com a tela"): quando o teclado fecha, o iPhone às vezes deixa
  420| // as barras de baixo no meio da tela. O app percebe o teclado, esconde a barra de abas enquanto ele está aberto e,
  421| // quando fecha, dá um "empurrãozinho" na rolagem pra ele recalcular o lugar das barras.
  422| export function ligarTeclado() {
  423|   const vv = window.visualViewport;
  424|   if (!vv) return;
  425|   let aberto = false;
  426|   const conferir = () => {
  427|     const agora = innerHeight - vv.height > 140;
  428|     if (agora === aberto) return;
  429|     aberto = agora;
  430|     document.body.classList.toggle('teclado', aberto);
  431|     if (!aberto) requestAnimationFrame(() => { window.scrollTo(window.scrollX, window.scrollY + 1); window.scrollTo(window.scrollX, window.scrollY - 1); });
  432|   };
  433|   vv.addEventListener('resize', conferir);
  434|   document.addEventListener('focusout', () => setTimeout(() => {
  435|     conferir();
  436|     if (!document.activeElement?.matches?.('input, textarea, select')) window.scrollTo(window.scrollX, window.scrollY);
  437|   }, 60));
  438| }
