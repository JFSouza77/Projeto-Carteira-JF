   66| async function lerCorpo(req) {
   67|   const tam = Number(req.headers.get('content-length') || 0);
   68|   if (tam > MAX_CORPO) throw new ErroHttp(413, 'Conteúdo grande demais.');
   69|   // Conta BYTES enquanto lê: para no limite, sem guardar o resto na memória.
   70|   const partes = [];
   71|   let bytes = 0;
   72|   if (req.body) {
   73|     const leitor = req.body.getReader();
   74|     for (;;) {
   75|       const { done, value } = await leitor.read();
   76|       if (done) break;
   77|       bytes += value.byteLength;
   78|       if (bytes > MAX_CORPO) {
   79|         await leitor.cancel().catch(() => {});
   80|         throw new ErroHttp(413, 'Conteúdo grande demais.');
   81|       }
   82|       partes.push(value);
   83|     }
   84|   }
   85|   const junto = new Uint8Array(bytes);
   86|   let pos = 0;
   87|   for (const p of partes) { junto.set(p, pos); pos += p.byteLength; }
   88|   let corpo;
   89|   try {
   90|     corpo = JSON.parse(new TextDecoder().decode(junto));
   91|   } catch {
   92|     throw new ErroHttp(400, 'JSON inválido.');
   93|   }
   94|   // JSON válido mas que não é objeto (null, número, lista): pedido mal formado (revisão 06/11).
   95|   if (!corpo || typeof corpo !== 'object' || Array.isArray(corpo)) throw new ErroHttp(400, 'O pedido precisa ser um objeto JSON.');
   96|   return corpo;
   97| }

  124|   // Sem KV não há como contar: as rotas pagas falham fechado em vez de rodar sem limite (revisão 06/11).
  125|   if (!env.CARTEIRA_KV) throw new ErroHttp(503, 'Servidor sem o armazenamento de limites (CARTEIRA_KV). A IA fica desligada até configurar.');

  129|   const kTot = `lim:${dia}:total`;

  135|   if (Number(nTot || 0) >= Number(env.IA_TETO_DIA || 120)) throw new ErroHttp(429, 'A IA descansou por hoje (teto do dia).');

  145| export async function sha256(t) {
  146|   const d = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(t));

  759| async function hashSenha(senha, salHex) {
  760|   const chave = await crypto.subtle.importKey('raw', new TextEncoder().encode(senha), 'PBKDF2', false, ['deriveBits']);
  761|   const bits = await crypto.subtle.deriveBits({ name: 'PBKDF2', hash: 'SHA-256', salt: deHex(salHex), iterations: ITERACOES }, chave, 256);
  762|   return hex(bits);
  763| }

  765| function igual(a, b) {
  766|   if (typeof a !== 'string' || typeof b !== 'string' || a.length !== b.length) return false;
  767|   let d = 0;
  768|   for (let i = 0; i < a.length; i++) d |= a.charCodeAt(i) ^ b.charCodeAt(i);
  769|   return d === 0;
  770| }
