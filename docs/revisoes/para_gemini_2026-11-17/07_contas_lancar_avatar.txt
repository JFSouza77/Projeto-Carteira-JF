  137| export async function registrarGasto({ amountCents, categoryId, note, date, accountId, imageId, compromissoId, rachaNomes = null, forma = null, parcelas = 1, limiteModo = 'total' }) {
  138|   const ofAntes = ofensivaAgora();
  139|   // escolheu um cartão sem dizer a forma: é crédito
  140|   if (!forma && accountId && state.accounts.find(a => a.id === accountId)?.tipo === 'cartao') forma = 'credito';
  141|   const expense = createExpense({ amountCents, categoryId, note, date, accountId, imageId, compromissoId, forma, parcelas, limiteModo }, { categories: state.categories });
  142|   if (rachaNomes?.length) expense.racha = dividirIgual(amountCents, rachaNomes);
  143|   const conta = accountId && state.accounts.find(a => a.id === accountId);
  144|   const cartao = conta?.tipo === 'cartao';
  145|   const antes = conta ? (cartao ? faturaDoCartao(comReforco(conta, state.accounts, state.expenses, state.incomes, state.transferencias), state.expenses, toISODate()).pct : accountBalance(conta, state.expenses, state.incomes, state.transferencias)) : null;
  146|   await db.put(db.STORES.expenses, expense);
  147|   avisarAntesDoSaldo(expense);
  148|   state.expenses.push(expense);
  149|   const events = detectExpenseEvents({ expenses: state.expenses, categories: state.categories, newExpense: expense });
  150|   if (conta && cartao) {
  151|     // passou de 80% ou de 100% do limite com esta compra: o C.A.O.S. avisa na hora
  152|     const f = faturaDoCartao(comReforco(conta, state.accounts, state.expenses, state.incomes, state.transferencias), state.expenses, toISODate());
  153|     if ((antes < 80 && f.pct >= 80) || (antes < 100 && f.pct >= 100)) {
  154|       events.push({ type: 'cartao_limite', accountName: f.nome, pct: f.pct, amountCents: f.usadoCents, leftCents: f.disponivelCents, budgetCents: f.limiteCents });
  155|     }
  156|   } else if (conta) {
  157|     const depois = accountBalance(conta, state.expenses, state.incomes, state.transferencias);
  158|     if (antes >= 0 && depois < 0) events.push({ type: 'conta_negativa', accountName: accountLabel(conta), balanceCents: depois });
  159|   }
  160|   if (expense.racha) {
  161|     const nomes = expense.racha.map(p => p.nome);
  162|     events.push({ type: 'racha_cadastrada', amountCents: valorProprio(expense), totalCents: expense.amountCents - valorProprio(expense), count: nomes.length,
  163|       quem: juntarNomes(nomes) });
  164|   }
  165|   events.push(...eventoOfensiva(ofAntes));
  166|   // guardar a cada compra (JF, 16/11, como no Mercado Pago): o cofrinho recebe o valor combinado, saindo da conta
  167|   // de origem. Fica ligado ao gasto (autoDe): apagar o gasto apaga o que foi guardado por causa dele.
  168|   const guardas = guardarACadaCompra(expense, state.accounts, state.expenses, state.incomes, state.transferencias);
  169|   if (guardas.length) {
  170|     const novas = guardas.map(gd => ({ ...createTransferencia({ deId: gd.deId, paraId: gd.cofrinhoId, amountCents: gd.amountCents, date: expense.date,
  171|       note: 'Guardado a cada compra' }, { accounts: state.accounts }), autoDe: expense.id }));
  172|     state.transferencias = [...state.transferencias, ...novas];
  173|     await db.setMeta('transferencias', state.transferencias);
  174|     const nomes = guardas.map(gd => `${formatBRL(gd.amountCents)} em ${accountLabel(state.accounts.find(a => a.id === gd.cofrinhoId))}`).join(' e ');
  175|     setTimeout(() => toast(`Guardei ${nomes}, como você combinou no banco.`), 1200);
  176|   }
  177|   state.viewMonth = expense.month;
  178|   renderizar();
  179|   return events;
  180| }

  206| export async function onSubmit(ev) {
  207|   ev.preventDefault();
  208|   const err = $('#form-error');
  209|   err.hidden = true;
  210|   try {
  211|     const amountCents = parseAmount($('#amount').value);
  212|     if (!amountCents) throw new Error('Digite um valor, tipo 25,90.');
  213|     // valor muito acima do normal (13/11): pergunta antes, pra um 124,50 não virar 12.450,00
  214|     const ehGasto = state.kind === 'gasto';
  215|     if (valorForaDoComum(amountCents, ehGasto ? state.expenses : state.incomes, { tipo: ehGasto ? 'gasto' : 'entrada' })
  216|       && !(await confirmarEspanto(amountCents, ehGasto))) { $('#amount').focus(); return; }
  217|     const base = { amountCents, note: $('#note').value, date: $('#date').value, accountId: $('#account-select').value || null };
  218|     let events;
  219|     if (state.kind === 'gasto') {
  220|       if (!state.selectedCategory) throw new Error('Escolha uma categoria.');
  221|       const rachaNomes = $('#racha-box').open ? nomesDoRacha() : null;
  222|       if ($('#racha-box').open && !rachaNomes.length) throw new Error('Escreva o nome de quem vai te pagar, ou feche o "Dividir com amigos".');
  223|       const forma = state.forma;
  224|       const parcelas = forma === 'credito' ? Number($('#parcelas-select').value) || 1 : 1;
  225|       const limiteModo = document.querySelector('input[name="limite-modo"]:checked')?.value || 'total';
  226|       if (forma === 'credito' && !base.accountId && state.accounts.some(a => a.tipo === 'cartao')) throw new Error('Qual cartão? Escolha ali em cima.');
  227|       events = await registrarGasto({ ...base, categoryId: state.selectedCategory, imageId: state.photo?.id, rachaNomes, forma, parcelas, limiteModo });

  236|       if (prevista && base.date < toISODate()) throw new Error('Entrada prevista é pra uma data que ainda vai chegar.');
  237|       events = await registrarEntrada({ ...base, sourceId: state.selectedSource, prevista });
  238|       $('#entrada-prevista').checked = false;
  239|       await limparFoto({ apagar: true });

  241|     $('#amount').value = '';
  242|     $('#note').value = '';
  243|     state.catSugerida = false;
  244|     state.selectedCategory = null; // o próximo lançamento começa livre pra o C.A.O.S. sugerir pela descrição

  249|   } catch (e) {
  250|     err.textContent = e.message || 'Não deu para salvar.';

   19|   if (!base) throw new Error('Configure o endereço do servidor em Ajustes → IA online.');
   20|   const ctrl = new AbortController();
   21|   const t = setTimeout(() => ctrl.abort(), TEMPO_MS);
   22|   try {
   23|     const r = await fetch(base.replace(/\/+$/, '') + rota, {
   24|       method: corpo ? 'POST' : 'GET',
   25|       headers: { ...(corpo ? { 'content-type': 'application/json' } : {}), ...(tokenCriador && rota.startsWith('/ia/') ? { authorization: `Bearer ${tokenCriador}` } : {}) },
   26|       body: corpo ? JSON.stringify(corpo) : undefined,
   27|       signal: ctrl.signal,
   28|     });
   29|     const j = await r.json().catch(() => ({}));
   30|     if (!r.ok) throw new Error(j.erro || `O servidor respondeu ${r.status}.`);
   31|     return j;
   32|   } catch (e) {
   33|     if (e.name === 'AbortError') throw new Error('O servidor demorou demais. Tenta de novo.');
   34|     if (e instanceof TypeError) throw new Error('Sem conexão com o servidor da IA.');
   35|     throw e;
   36|   } finally {
   37|     clearTimeout(t);
   38|   }
